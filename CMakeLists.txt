cmake_minimum_required(VERSION 3.24)
project(SkGPUTape VERSION 0.1.0 LANGUAGES CXX)
option(SKGPUTAPE_CORE_TESTS_ONLY "Build portable discovery tests without Skyrim" OFF)
if(SKGPUTAPE_CORE_TESTS_ONLY)
  enable_testing()
  add_executable(discovery-tests tests/AssetDiscoveryTests.cpp src/AssetDiscovery.cpp)
  target_include_directories(discovery-tests PRIVATE src)
  target_compile_features(discovery-tests PRIVATE cxx_std_20)
  add_test(NAME discovery COMMAND discovery-tests)
  return()
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(CommonLibSSE
  GIT_REPOSITORY https://github.com/CharmedBaryon/CommonLibSSE-NG.git
  GIT_TAG b93280e832f263dbef44e44cbe2936622a02f91a)
FetchContent_MakeAvailable(CommonLibSSE)
include(${commonlibsse_SOURCE_DIR}/cmake/CommonLibSSE.cmake)
add_commonlibsse_plugin(SkGPUTape AUTHOR "ArcaneArts" VERSION ${PROJECT_VERSION}
  SOURCES src/main.cpp src/Config.cpp src/AssetDiscovery.cpp src/GPUWarmupManager.cpp)
target_compile_features(SkGPUTape PRIVATE cxx_std_23)
target_compile_options(SkGPUTape PRIVATE /W4 /permissive- /EHsc)
target_precompile_headers(SkGPUTape PRIVATE src/PCH.h)
set_source_files_properties(src/Config.cpp PROPERTIES SKIP_PRECOMPILE_HEADERS ON)
install(TARGETS SkGPUTape RUNTIME DESTINATION SKSE/Plugins COMPONENT SkGPUTape)
install(FILES dist/SkGPUTape.ini dist/SkGPUTapeModels.txt DESTINATION SKSE/Plugins COMPONENT SkGPUTape)
install(FILES README.md docs/TESTING.md docs/API_RESEARCH.md DESTINATION SkGPUTape COMPONENT SkGPUTape)
set(CPACK_INSTALL_CMAKE_PROJECTS "${CMAKE_BINARY_DIR};${PROJECT_NAME};SkGPUTape;/")
set(CPACK_GENERATOR ZIP)
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
set(CPACK_PACKAGE_FILE_NAME "SkGPUTape-${PROJECT_VERSION}-win64")
include(CPack)
